"""Diagnostic only: compare fixed-link guitar with an equivalent compound body.

No production module is modified. Inspect must run in its own process; replay
uses a private copied reset contract and retains canonical106-row tape output.
"""
import argparse
import copy
import hashlib
import json
import os
from pathlib import Path
import sys
import time
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))


def write(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)


def pose(row):
    import numpy as np
    from scipy.spatial.transform import Rotation
    return (np.array([row['pose']['p'][a] for a in 'xyz']),
            Rotation.from_quat([row['pose']['r'][a] for a in 'xyzw']))


def compound_contract(contract, asset, out):
    """Aggregate the actual pinned body inertias, preserving their COM frames."""
    import numpy as np
    initial, saved = contract['initial'], contract['state']
    rows = initial['guitar_properties']['applied']
    p0, r0 = pose(saved['guitar_body_states'][0])
    transforms = [pose(row) for row in saved['guitar_body_states']]
    centers, rotations = [], []
    for row, (p, r) in zip(rows, transforms):
        centers.append(r0.inv().apply(p + r.apply(row['COM_body_m']) - p0))
        rotations.append((r0.inv() * r).as_matrix())
    masses = np.array([row['mass_kg'] for row in rows])
    centers = np.array(centers)
    mass = masses.sum()
    center = (masses[:, None] * centers).sum(0) / mass
    inertia = np.zeros((3, 3))
    for m, c, rotation, row in zip(masses, centers, rotations, rows):
        offset = c - center
        inertia += rotation @ np.array(row['inertia_COM_kg_m2']) @ rotation.T
        inertia += m * (np.dot(offset, offset) * np.eye(3) - np.outer(offset, offset))
    merged = dict(name=rows[0]['name'], mass_kg=float(mass), COM_body_m=center.tolist(),
                  inertia_COM_kg_m2=inertia.tolist())
    shape_rows = [row for name in initial['guitar_body_names']
                  for row in initial['actual_shape_properties']['guitar'][name]]
    adapted = copy.deepcopy(contract)
    ai, state = adapted['initial'], adapted['state']
    ai['guitar_body_names'] = [merged['name']]
    ai['guitar_properties']['applied'] = [merged]
    ai['actual_shape_properties']['guitar'] = {merged['name']: shape_rows}
    for loaded in ai['loaded_assets']:
        if not loaded['fixed_base']:
            loaded['path'] = str(asset)
    state['guitar_body_states'] = [copy.deepcopy(saved['guitar_body_states'][0])]
    # Rigid-body linear velocity is at COM. The new COM differs from the old root.
    root = state['guitar_body_states'][0]
    v = np.array([root['vel']['linear'][a] for a in 'xyz'])
    w = np.array([root['vel']['angular'][a] for a in 'xyz'])
    v += np.cross(w, r0.apply(center - np.array(rows[0]['COM_body_m'])))
    root['vel']['linear'] = dict(zip('xyz', map(float, v)))
    write(out / 'compound_properties.json', dict(aggregate=merged, source_bodies=rows,
          source_shape_count=len(shape_rows), initial_velocity_COM_correction=v.tolist(),
          aggregate_inertia_eigenvalues=np.linalg.eigvalsh(inertia).tolist()))
    return adapted


def flatten_asset(source, original, destination):
    """Flatten MJCF body transforms, retaining every visual/collision geom.

    The pinned guitar uses pos/quat only; fail on additional orientation forms
    rather than silently changing geometry. Geom ordering follows native bodies.
    """
    import numpy as np
    from scipy.spatial.transform import Rotation
    tree = ET.parse(source)
    root = tree.getroot()
    compiler = root.find('compiler')
    meshdir = source.parent / compiler.get('meshdir', '')
    for mesh in root.findall('./asset/mesh'):
        mesh.set('file', os.path.relpath((meshdir / mesh.attrib['file']).resolve(), destination.parent))
    compiler.set('meshdir', '.')
    guitar = root.find('./worldbody/body')
    if list(guitar.iter('joint')) or list(guitar.iter('freejoint')):
        raise ValueError('Flattening is only equivalent for a zero-joint guitar')
    geoms, frames = {}, {}

    def walk(body, p, r):
        if any(k in body.attrib for k in ('euler', 'axisangle', 'xyaxes', 'zaxis')):
            raise ValueError('Unsupported body orientation')
        local_p = np.fromstring(body.get('pos', '0 0 0'), sep=' ')
        wxyz = np.fromstring(body.get('quat', '1 0 0 0'), sep=' ')
        local_r = Rotation.from_quat(wxyz[[1, 2, 3, 0]])
        p, r = p + r.apply(local_p), r * local_r
        frames[body.attrib['name']] = (p, r)
        direct = []
        for geom in body.findall('geom'):
            if any(k in geom.attrib for k in ('euler', 'axisangle', 'xyaxes', 'zaxis', 'fromto')):
                raise ValueError('Unsupported geom orientation')
            geom = copy.deepcopy(geom)
            gp = np.fromstring(geom.get('pos', '0 0 0'), sep=' ')
            gq = np.fromstring(geom.get('quat', '1 0 0 0'), sep=' ')
            gr = r * Rotation.from_quat(gq[[1, 2, 3, 0]])
            geom.set('pos', ' '.join(format(x, '.17g') for x in p + r.apply(gp)))
            xyzw = gr.as_quat()
            geom.set('quat', ' '.join(format(x, '.17g') for x in xyzw[[3, 0, 1, 2]]))
            direct.append(geom)
        geoms[body.attrib['name']] = direct
        for child in body.findall('body'):
            walk(child, p, r)

    # The root is identity in the pinned asset. Keep this explicit: actor poses
    # and aggregate inertia are expressed in that exact root coordinate system.
    if guitar.get('pos', '0 0 0') != '0 0 0' or guitar.get('quat', '1 0 0 0') != '1 0 0 0':
        raise ValueError('Pinned guitar root must have identity local transform')
    walk(guitar, np.zeros(3), Rotation.identity())
    for child in list(guitar):
        guitar.remove(child)
    initial = original['initial']
    collision_count = 0
    for name in initial['guitar_body_names']:
        enabled = [g for g in geoms[name] if g.get('contype', '1') != '0']
        expected = initial['actual_shape_properties']['guitar'][name]
        if len(enabled) != len(expected):
            raise ValueError('Source collision-geom/native shape count differs for ' + name)
        collision_count += len(enabled)
        for geom in geoms[name]:
            guitar.append(geom)
    tree.write(destination, encoding='utf-8', xml_declaration=True)
    return dict(source=str(source), path=str(destination), bodies_before=len(frames),
                bodies_after=1, all_geoms=len(list(guitar.iter('geom'))), collision_geoms=collision_count,
                source_body_order=initial['guitar_body_names'],
                source_sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
                generated_sha256=hashlib.sha256(destination.read_bytes()).hexdigest())


def inspect_asset(contract, source, collapse, out):
    from isaacgym import gymapi
    gym = gymapi.acquire_gym()
    params = gymapi.SimParams()
    params.use_gpu_pipeline = params.physx.use_gpu = True
    sim = gym.create_sim(0, -1, gymapi.SIM_PHYSX, params)
    if sim is None:
        raise RuntimeError('Asset-inspection GPU simulator failed')
    try:
        options = gymapi.AssetOptions()
        options.fix_base_link = options.disable_gravity = False
        options.override_com = options.override_inertia = True
        options.collapse_fixed_joints = collapse
        asset = gym.load_asset(sim, str(source.parent), source.name, options)
        if asset is None:
            raise RuntimeError('Asset import failed')
        names = list(gym.get_asset_rigid_body_names(asset))
        counts = dict(body_count=gym.get_asset_rigid_body_count(asset),
                      shape_count=gym.get_asset_rigid_shape_count(asset),
                      joint_count=gym.get_asset_joint_count(asset),
                      dof_count=gym.get_asset_dof_count(asset), body_names=names,
                      joint_names=list(gym.get_asset_joint_names(asset)),
                      requested_collapse_fixed_joints=collapse, source=str(source))
        write(out / 'asset_inspection.json', counts)
        print(json.dumps(counts), flush=True)
    finally:
        gym.destroy_sim(sim)


def canonical_bodies(scene, original):
    """Reconstruct fixed marker links solely for original106-row comparisons."""
    import numpy as np
    import torch
    from guitar_learning.gpu.geometry import multiply, apply as rotate
    saved = original['state']['guitar_body_states']
    p0, r0 = pose(saved[0])
    root_q = np.array([saved[0]['pose']['r'][a] for a in 'xyzw'])
    local_p, local_q = [], []
    for record in saved:
        p, r = pose(record)
        local_p.append(r0.inv().apply(p - p0))
        relative_q = (r0.inv() * r).as_quat()
        composed_q = np.r_[root_q[3] * relative_q[:3] + relative_q[3] * root_q[:3] +
                           np.cross(root_q[:3], relative_q[:3]),
                           root_q[3] * relative_q[3] - np.dot(root_q[:3], relative_q[:3])]
        expected_q = np.array([record['pose']['r'][a] for a in 'xyzw'])
        local_q.append(relative_q if np.dot(composed_q, expected_q) >= 0 else -relative_q)
    offsets = torch.tensor(np.array(local_p), dtype=torch.float32, device=scene.device)
    rotations = torch.tensor(np.array(local_q), dtype=torch.float32, device=scene.device)
    native = scene.body_state
    h = len(scene.human_bodies)
    if len(scene.guitar_bodies) != 1:
        return native
    base = native[:, h:h+1]
    guitar = base.expand(-1, len(saved), -1).clone()
    guitar[..., :3] = base[..., :3] + rotate(base[..., 3:7], offsets[None])
    guitar[..., 3:7] = multiply(base[..., 3:7], rotations[None])
    old_com = torch.tensor([r['COM_body_m'] for r in original['initial']['guitar_properties']['applied']],
                           dtype=torch.float32, device=scene.device)
    compound_com = scene.com[h]
    com_delta = (guitar[..., :3] + rotate(guitar[..., 3:7], old_com[None]) -
                 (base[..., :3] + rotate(base[..., 3:7], compound_com)))
    guitar[..., 7:10] = base[..., 7:10] + torch.cross(base[..., 10:13].expand_as(com_delta), com_delta, dim=-1)
    return torch.cat((native[:, :h], guitar, native[:, -1:]), dim=1)


def replay(args, original, adapted, source):
    import numpy as np
    import torch
    from isaacgym import gymapi
    from guitar_learning.gpu.scene import GpuScene
    from guitar_learning.gpu.geometry import conjugate, multiply, rotvec
    import guitar_learning.runtime.prepared_reset_v1 as prepared
    # Private process-local patches only; no source or pinned evidence mutation.
    prepared.load_prepared_reset_contract = lambda: copy.deepcopy(adapted)
    if args.representation == 'collapse':
        acquire = gymapi.acquire_gym
        class GymProxy:
            def __init__(self, gym):
                self.gym = gym
            def __getattr__(self, name):
                return getattr(self.gym, name)
            def load_asset(self, sim, folder, name, options):
                if (Path(folder) / name).resolve() == source.resolve():
                    options.collapse_fixed_joints = True
                return self.gym.load_asset(sim, folder, name, options)
        gymapi.acquire_gym = lambda: GymProxy(acquire())
    tape = np.load(args.tape)
    scene = GpuScene(args.envs, 'left', out=args.out / 'scene')
    ids = [i for i in (0, 1, 33, 64, 512, 1023) if i < args.envs]
    selected = torch.tensor(ids, device=scene.device)
    initial = canonical_bodies(scene, original)
    h = len(scene.human_bodies)
    feet = [scene.human_bodies.index(s + '_Ankle') for s in ('L', 'R')]
    body, q, qd = [initial[selected].cpu().numpy()], [scene.q[selected].cpu().numpy()], [scene.qd[selected].cpu().numpy()]
    first = torch.full((args.envs,), -1, device=scene.device, dtype=torch.long)
    rows, failure = [], None
    started = time.monotonic()
    try:
        with torch.no_grad():
            for step in range(min(args.steps, len(tape['efforts']))):
                target = torch.as_tensor(tape['targets'][step], device=scene.device).expand(args.envs, -1)
                effort = torch.as_tensor(tape['efforts'][step], device=scene.device).expand(args.envs, -1)
                snapshot = scene.step(target, effort)
                b = canonical_bodies(scene, original)
                if not bool(torch.isfinite(b).all() & torch.isfinite(scene.qd).all()):
                    raise FloatingPointError('Nonfinite native physics response')
                delta = torch.linalg.vector_norm(b[:, h, :3] - initial[:, h, :3], dim=-1)
                angle = torch.linalg.vector_norm(rotvec(multiply(conjugate(initial[:, h, 3:7]), b[:, h, 3:7])), dim=-1)
                foot = torch.linalg.vector_norm(b[:, feet, :3] - initial[:, feet, :3], dim=-1).amax(-1)
                speed = scene.qd.abs().amax(-1)
                bad = (delta > .08) | (angle > .35) | (foot > .04) | (speed > 50)
                first = torch.where((first < 0) & bad, step + 1, first)
                body.append(b[selected].cpu().numpy()); q.append(scene.q[selected].cpu().numpy()); qd.append(scene.qd[selected].cpu().numpy())
                rows.append(dict(step=step + 1, ever_failed=int((first >= 0).sum()), current_failed=int(bad.sum()),
                    displacement_m=delta[selected].tolist(), rotation_rad=angle[selected].tolist(),
                    foot_delta_m=foot[selected].tolist(), speed_rad_s=speed[selected].tolist()))
    except BaseException as error:
        failure = dict(type=type(error).__name__, message=str(error))
        raise
    finally:
        scene.close()
        np.savez_compressed(args.out / 'tape.npz', targets=tape['targets'][:len(rows)],
            efforts=tape['efforts'][:len(rows)], body=np.stack(body), q=np.stack(q), qd=np.stack(qd))
        write(args.out / 'report.json', dict(completed=failure is None, failure=failure,
            representation=args.representation, num_envs=args.envs, env_ids=ids, steps=len(rows),
            first_failure_physics_steps=first.cpu().tolist(), rows=rows,
            elapsed_s=time.monotonic() - started, scene=scene.diagnostics,
            body_output='Canonical106: guitar marker poses reconstructed rigidly from compound root; marker velocities at original COM',
            replay=str(args.tape.resolve()), replay_sha256=hashlib.sha256(args.tape.read_bytes()).hexdigest()))
        print('completed', args.out, 'steps', len(rows), 'failed', int((first >= 0).sum()), flush=True)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--out', type=Path, required=True)
    parser.add_argument('--mode', choices=('inspect', 'replay', 'generate'), default='inspect')
    parser.add_argument('--representation', choices=('original', 'collapse', 'flat'), default='collapse')
    parser.add_argument('--envs', type=int, default=1)
    parser.add_argument('--steps', type=int, default=128)
    parser.add_argument('--tape', type=Path)
    args = parser.parse_args()
    if args.mode == 'replay' and args.tape is None:
        parser.error('--tape required for replay')
    import isaacgym  # Must precede Torch imported transitively by prepared reset.
    from guitar_learning.runtime.prepared_reset_v1 import load_prepared_reset_contract
    from guitar_learning.native.paths import resolve_input_path
    args.out.mkdir(parents=True, exist_ok=False)
    contract = load_prepared_reset_contract()
    source = resolve_input_path(next(r['path'] for r in contract['initial']['loaded_assets'] if not r['fixed_base']))
    if args.representation == 'flat':
        destination = args.out / 'guitar_compound.xml'
        write(args.out / 'geometry.json', flatten_asset(source, contract, destination))
        source = destination
    if args.mode == 'generate':
        if args.representation != 'flat':
            parser.error('generate mode requires flat representation')
        compound_contract(contract, source, args.out)
    elif args.mode == 'inspect':
        inspect_asset(contract, source, args.representation == 'collapse', args.out)
    else:
        adapted = (contract if args.representation == 'original' else
                   compound_contract(contract, source, args.out))
        replay(args, contract, adapted, source)


if __name__ == '__main__':
    main()
