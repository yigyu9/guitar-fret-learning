import argparse,csv,json,os,signal,subprocess,time
from pathlib import Path
BASE=Path('/home/ajou/yigyu/3/docs/2026-10-20/guitar_learning_large_gpu_env_benchmark_v1')
p=argparse.ArgumentParser();p.add_argument('--name',required=True);p.add_argument('--num-envs',type=int,choices=(2,4,8),required=True);p.add_argument('--updates',type=int,default=1);p.add_argument('--horizon',type=int,default=1969);p.add_argument('--resume',type=Path);a=p.parse_args()
assert a.name in ('n2','n4','n8','n8_repeat') and a.updates in (1,2) and a.horizon in (1969,256)
out=BASE/a.name
assert not out.exists()
env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1',TORCH_EXTENSIONS_DIR='/tmp/guitar_structure_torch_extensions',OMP_NUM_THREADS='1',OPENBLAS_NUM_THREADS='1',MKL_NUM_THREADS='1',PYTHONPATH='/home/ajou/yigyu/3/isaacgym/python:/home/ajou/yigyu/3',LD_LIBRARY_PATH='/home/ajou/miniconda3/envs/rl38/lib',PATH='/home/ajou/miniconda3/envs/rl38/bin:'+os.environ['PATH'])
command=['/home/ajou/miniconda3/envs/rl38/bin/python','-B','-u','-m','guitar_learning.train','--hand','left','--song','05_BN2-131-B_solo','--size','large','--num-envs',str(a.num_envs),'--updates',str(a.updates),'--horizon',str(a.horizon),'--seed','7','--device','cuda:0','--out',str(out)]
if a.resume:command+=['--resume',str(a.resume.resolve())]
started=time.monotonic();rows=[];interrupted_at=None;guard=None
with (BASE/(a.name+'_console.log')).open('x') as log,(BASE/(a.name+'_gpu_samples.jsonl')).open('x') as samples:
 child=subprocess.Popen(command,cwd='/home/ajou/yigyu/3',env=env,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
 try:
  while True:
   probe=subprocess.run(['nvidia-smi','-i','0','--query-gpu=timestamp,index,name,memory.total,memory.used,memory.free,utilization.gpu','--format=csv,noheader,nounits'],capture_output=True,text=True,timeout=10)
   row=dict(elapsed_s=time.monotonic()-started,returncode=probe.returncode,stdout=probe.stdout,stderr=probe.stderr)
   if probe.returncode==0:
    values=next(csv.reader(probe.stdout.splitlines()));row.update(timestamp=values[0].strip(),index=int(values[1]),name=values[2].strip(),total_mib=float(values[3]),used_mib=float(values[4]),free_mib=float(values[5]),gpu_utilization_percent=float(values[6]))
   memory={s.split(':')[0]:int(s.split()[1])*1024 for s in Path('/proc/meminfo').read_text().splitlines() if s.startswith(('MemAvailable:','SwapFree:'))};row.update(memory)
   status=out/'status.json'
   if status.exists():
    state=json.loads(status.read_text());row['phase']=state['phase'];row['ticks']=state.get('ticks');row['update_index']=state.get('update_index')
   rows.append(row);samples.write(json.dumps(row)+'\n');samples.flush()
   risk=probe.returncode!=0 or row.get('free_mib',0)<1024 or row['MemAvailable']<6*2**30
   if risk and interrupted_at is None:
    guard=row;interrupted_at=time.monotonic();child.send_signal(signal.SIGINT)
   if interrupted_at is not None and time.monotonic()-interrupted_at>60 and child.poll() is None:os.killpg(child.pid,signal.SIGTERM)
   if child.poll() is not None:break
   time.sleep(2)
 except BaseException:
  if child.poll() is None:child.send_signal(signal.SIGINT)
  try:child.wait(timeout=60)
  except subprocess.TimeoutExpired:os.killpg(child.pid,signal.SIGTERM);child.wait(timeout=20)
  raise
 code=child.wait()
valid=[r for r in rows if r['returncode']==0]
result=dict(completed=code==0 and guard is None and len(valid)==len(rows),command=command,returncode=code,elapsed_s=time.monotonic()-started,guard_trigger=guard,samples=len(rows),gpu_sample_peak_used_mib=max(r['used_mib'] for r in valid) if valid else None,gpu_sample_min_free_mib=min(r['free_mib'] for r in valid) if valid else None,host_sample_min_available_gib=min(r['MemAvailable'] for r in rows)/2**30,scope='globalGPU nvidia-smi approximately2s samples plus native CLI; Torch peak allocated/reserved is in run report')
(BASE/(a.name+'_supervisor.json')).write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result),flush=True)
raise SystemExit(0 if result['completed'] else 1)
