class AuditedScene(base):
    def __init__(self,*a,**kw):
        super().__init__(*a,**kw)
        actor_bad=[];dof_bad=[];body_bad=[]
        for i,(env,actors) in enumerate(zip(self.envs,self.actors)):
            for j,actor in enumerate(actors):
                index=self.gym.get_actor_index(env,actor,gymapi.DOMAIN_SIM)
                if index!=3*i+j:actor_bad.append([i,j,index])
            for j in range(105):
                index=self.gym.get_actor_dof_index(env,actors[0],j,gymapi.DOMAIN_SIM)
                if index!=105*i+j:dof_bad.append([i,j,index])
            offset=0
            for actor,n in zip(actors,(len(self.human_bodies),len(self.guitar_bodies),1)):
                for j in range(n):
                    index=self.gym.get_actor_rigid_body_index(env,actor,j,gymapi.DOMAIN_SIM)
                    if index!=self.body_count*i+offset+j:body_bad.append([i,offset+j,index])
                offset+=n
        self.diagnostic_contact_max = 0.
        rest_values=[];nonuniform=0;shapes0=None
        for env,actors in zip(self.envs,self.actors):
            shapes={}
            for label,actor in zip(('human','guitar','chair'),actors):
                values=self.gym.get_actor_rigid_shape_properties(env,actor)
                shapes[label]=[{key:getattr(v,key) for key in ('filter','friction','rolling_friction','torsion_friction','restitution','compliance','thickness','contact_offset','rest_offset') if hasattr(v,key)} for v in values]
                rest_values.extend(v.rest_offset for v in values if hasattr(v,'rest_offset'))
            if shapes0 is None:shapes0=shapes
            elif shapes!=shapes0:nonuniform+=1
        with (args.out/'full_shape_env0.json').open('x') as f:json.dump(shapes0,f,indent=2)
        with (args.out/'shape_summary.json').open('x') as f:json.dump(dict(nonuniform_envs=nonuniform,shape_rest_min=min(rest_values) if rest_values else None,shape_rest_max=max(rest_values) if rest_values else None,shape_rest_samples=len(rest_values)),f,indent=2)
        actual=self.gym.get_sim_params(self.sim)
        with (args.out/'full_physx.json').open('x') as f:json.dump({key:getattr(actual.physx,key) for key in dir(actual.physx) if not key.startswith('_') and isinstance(getattr(actual.physx,key),(int,float,bool))},f,indent=2)
        with (args.out/'env_origins.json').open('x') as f:json.dump(self.origins.tolist(),f)

        data=dict(actor_index_errors=actor_bad,dof_index_errors=dof_bad,body_index_errors=body_bad,
            sim_actor_count=self.gym.get_sim_actor_count(self.sim),sim_dof_count=self.gym.get_sim_dof_count(self.sim),
            sim_body_count=self.gym.get_sim_rigid_body_count(self.sim),actual_flags=self.diagnostics,
            variant=args.variant,source_sha256=hashlib.sha256(source.encode()).hexdigest())
        with (args.out/'native_index_audit.json').open('x') as f:json.dump(data,f,indent=2)
    def step(self, *a, **kw):
        sample = super().step(*a, **kw)
        if args.variant == 'no_contact':
            self.diagnostic_contact_max = max(self.diagnostic_contact_max, float(sample['contact_force'].abs().max()))
            if self.diagnostic_contact_max != 0.:
                raise RuntimeError('No-contact diagnostic unexpectedly has contact forces')
        if args.trace_contacts and self.physics_steps <= 8:
            from guitar_learning.native.native_contacts import plain
            if self.diagnostics['use_gpu_pipeline']:
                raise RuntimeError('Contact trace requires CPU tensor pipeline')
            rows=[]
            for index in (0,):
                rows.append(dict(env=index,contacts=[plain(v) for v in self.gym.get_env_rigid_contacts(self.envs[index])]))
            with (args.out/'contacts.jsonl').open('a') as f:f.write(json.dumps(dict(step=self.physics_steps,rows=rows),allow_nan=False)+'\n')
        return sample
